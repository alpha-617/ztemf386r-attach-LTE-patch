for i in 1 2 3 4 5; do
    manufacturer=$(gcom -d "$device" -s /etc/gcom/getcardinfo.gcom 2>/dev/null | awk '
        NF && $0 !~ /AT\+CGMI/ {
            sub(/\+CGMI:[[:space:]]*/,"");
            gsub(/"/,"");
            split($0,a," ");
            print tolower(a[1]);
            exit;
        }')

    case "$manufacturer" in
        zte|huawei|quectel|fibocom)
            break
            ;;
        *)
            manufacturer=""
            ;;
    esac

    sleep 2
done
